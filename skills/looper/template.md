  mode: hard                  # hard (потолок, дальше throw) | soft (порог-предупреждение)
  cap: ""                     # напр. "+300k"; сворачиваться по остатку

  who: codex                  # self | codex | other-model — НЕ исполнитель
  how: ""                     # оценка по rubric | N скептиков-голосов | refute-mode
  pass_threshold: ""          # напр. "≥2 из 3 за"

  where: ""                   # файл/память/inbox — переживает итерацию
